{{- /* ZISAI responsive thumbnail. Requires static/images -> assets/images mount. */ -}} {{- $src := .src -}} {{- $alt := .alt | default "" -}} {{- $loading := .loading | default "lazy" -}} {{- $sizes := .sizes | default "100vw" -}} {{- $widths := .widths | default (slice 320 480 640) -}} {{- $page := .page -}} {{- $originalURL := $src | relURL -}} {{- $resourcePath := strings.TrimPrefix "/" $src -}} {{- $res := resources.Get $resourcePath -}} {{- if and (not $res) $page -}} {{- $res = $page.Resources.GetMatch (path.Base $resourcePath) -}} {{- end -}} {{- $srcset := slice -}} {{- $selectedURL := $originalURL -}} {{- $processable := false -}} {{- with $res -}} {{- $processable = reflect.IsImageResourceProcessable . -}} {{- end -}} {{- if $processable -}} {{- $failed := false -}} {{- range $width := $widths -}} {{- if and (not $failed) (ge $res.Width $width) -}} {{- $attempt := try ($res.Resize (printf "%dx webp q82" $width)) -}} {{- if $attempt.Err -}} {{- warnf "ZISAI thumbnail resize skipped: %s (%v)" $src $attempt.Err -}} {{- $failed = true -}} {{- else -}} {{- $thumb := $attempt.Value -}} {{- $srcset = $srcset | append (printf "%s %dw" $thumb.RelPermalink $thumb.Width) -}} {{- if eq $selectedURL $originalURL -}} {{- $selectedURL = $thumb.RelPermalink -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- if $failed -}} {{- $srcset = slice -}} {{- $selectedURL = $originalURL -}} {{- end -}} {{- end -}}
